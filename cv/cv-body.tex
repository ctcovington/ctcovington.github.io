% --- IMPROVED CV ENTRY COMMAND ---
% Usage: \cventry{Date}{Title}{Institution}{Description}
% Fix: Checks if Description (#4) is empty.
% If empty, it suppresses the extra newline to avoid uneven spacing.
\newcommand{\cventry}[4]{
  \noindent\textbf{#2} \hfill #1 \\   % Bold Title ... Date
  \textit{#3}%                        % Italic Institution
  \ifstrempty{#4}{}{ \\ #4 }%         % Only add newline if Description exists
  \par\vspace{0.5em}                  % Consistent spacing after every entry
}

% Usage: \cventryone{Date}{Title}{Institution}{Description}
% Places Title and Institution on the same line.
\newcommand{\cventryone}[4]{
  \noindent\textbf{#2}, \textit{#3} \hfill #1 % Title, Inst ... Date
  \ifstrempty{#4}{}{ \\ #4 }%  Adds description on next line ONLY if it exists
  \par\vspace{2pt} % Tight spacing
}
